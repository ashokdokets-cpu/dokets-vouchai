@echo off
echo ============================================
echo    VouchAI - Windows Setup Script
echo ============================================
echo.

:: Check Node.js
echo [1/6] Checking Node.js installation...
node --version >nul 2>&1
if %errorlevel% neq 0 (
    echo ❌ Node.js is not installed!
    echo Please install Node.js from https://nodejs.org/
    pause
    exit /b 1
)
echo ✅ Node.js found
echo.

:: Check Python
echo [2/6] Checking Python installation...
python --version >nul 2>&1
if %errorlevel% neq 0 (
    echo ❌ Python is not installed!
    echo Please install Python from https://python.org/
    pause
    exit /b 1
)
echo ✅ Python found
echo.

:: Install root dependencies
echo [3/6] Installing root dependencies...
call npm install --legacy-peer-deps
if %errorlevel% neq 0 (
    echo ❌ Failed to install root dependencies
    pause
    exit /b 1
)
echo ✅ Root dependencies installed
echo.

:: Setup API package
echo [4/6] Setting up API package...
cd packages\api
call npm init -y
call npm install express cors helmet compression morgan dotenv
call npm install @prisma/client prisma
call npm install razorpay @paypal/checkout-server-sdk stripe
call npm install twilio ioredis bull jsonwebtoken bcryptjs
call npm install openai axios sharp multer uuid
call npm install socket.io ws
call npm install -D typescript @types/node @types/express ts-node nodemon
call npm install @vouchai/shared@file:../shared
echo ✅ API package setup complete
cd ..\..

:: Setup Web package
echo [5/6] Setting up Web package...
cd packages\web
call npm init -y
call npm install next@latest react@latest react-dom@latest
call npm install tailwindcss @tailwindcss/forms @tailwindcss/typography postcss autoprefixer
call npm install axios socket.io-client framer-motion
call npm install lucide-react @radix-ui/react-dialog @radix-ui/react-dropdown-menu
call npm install react-hook-form zod @hookform/resolvers
call npm install @vouchai/shared@file:../shared
call npm install -D typescript @types/node @types/react @types/react-dom
echo ✅ Web package setup complete
cd ..\..

:: Setup WhatsApp Bot
echo [6/6] Setting up WhatsApp Bot...
cd packages\whatsapp-bot
call npm init -y
call npm install twilio express axios dotenv
call npm install -D typescript @types/node ts-node
echo ✅ WhatsApp Bot package setup complete
cd ..\..

echo.
echo ============================================
echo    ✅ VouchAI setup complete!
echo ============================================
echo.
echo Next steps:
echo 1. Update .env files with your API keys
echo 2. Install PostgreSQL and Redis
echo 3. Run: cd packages\api ^&^& npx prisma migrate dev
echo 4. Run: start-dev.bat to start development
echo.
pause
